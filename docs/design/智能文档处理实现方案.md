# 智能文档处理系统实现方案

## 一、技术架构

### 1.1 核心依赖配置

在父 `pom.xml` 添加版本管理：

```xml

<properties>

    <poi.version>5.2.5</poi.version>

    <pdf-box.version>3.0.1</pdf-box.version>

    <tess4j.version>5.9.0</tess4j.version>

    <jodconverter.version>4.4.6</jodconverter.version>

    <thumbnailator.version>0.4.19</thumbnailator.version>

</properties>


<dependencyManagement>

    <dependencies>

        <!-- Apache POI (Excel/Word/PPT处理) -->

        <dependency>

            <groupId>org.apache.poi</groupId>

            <artifactId>poi-ooxml</artifactId>

            <version>${poi.version}</version>

        </dependency>

      

        <!-- PDF 处理 -->

        <dependency>

            <groupId>org.apache.pdfbox</groupId>

            <artifactId>pdfbox</artifactId>

            <version>${pdf-box.version}</version>

        </dependency>

      

        <!-- OCR 文字识别 -->

        <dependency>

            <groupId>net.sourceforge.tess4j</groupId>

            <artifactId>tess4j</artifactId>

            <version>${tess4j.version}</version>

        </dependency>

      

        <!-- 文档格式转换（LibreOffice） -->

        <dependency>

            <groupId>org.jodconverter</groupId>

            <artifactId>jodconverter-spring-boot-starter</artifactId>

            <version>${jodconverter.version}</version>

        </dependency>

      

        <dependency>

            <groupId>org.jodconverter</groupId>

            <artifactId>jodconverter-local</artifactId>

            <version>${jodconverter.version}</version>

        </dependency>

      

        <!-- 图片处理 -->

        <dependency>

            <groupId>net.coobird</groupId>

            <artifactId>thumbnailator</artifactId>

            <version>${thumbnailator.version}</version>

        </dependency>

    </dependencies>

</dependencyManagement>

```

```java
importjava.io.File;

importjava.nio.file.Paths;

/**

 * 文档预览服务

 */

@Slf4j

@Service

@RequiredArgsConstructor

publicclassDocumentPreviewService {

  

    privatefinalDocumentConvertServiceconvertService;

  

    @Value("${document.preview.cache-path:/data/preview}")

    privateStringpreviewCachePath;

  

    /**

     * 生成PDF缩略图（第一页）

     */

    publicString generatePdfThumbnail(StringpdfPath,intwidth,intheight)throwsIOException{

        try (PDDocumentdocument = PDDocument.load(newFile(pdfPath))) {

            PDFRendererrenderer = newPDFRenderer(document);

          

            // 渲染第一页

            BufferedImageimage = renderer.renderImageWithDPI(0,150,ImageType.RGB);

          

            // 生成缩略图

            StringthumbnailDir = previewCachePath +"/thumbnails/";

            FileUtil.mkdir(thumbnailDir);

            StringthumbnailPath = thumbnailDir +System.currentTimeMillis() +".jpg";

          

            // 缩放图片

            BufferedImagescaledImage = net.coobird.thumbnailator.Thumbnails.of(image)

                .size(width, height)

                .asBufferedImage();

          

            ImageIO.write(scaledImage,"JPEG",newFile(thumbnailPath));

          

            log.info("PDF缩略图生成成功：{}", thumbnailPath);

            return thumbnailPath;

        }

    }

  

    /**

     * PDF转图片（所有页）

     */

    publicList<String> convertPdfToImages(StringpdfPath)throwsIOException{

        List<String> imagePaths = newArrayList<>();

      

        try (PDDocumentdocument = PDDocument.load(newFile(pdfPath))) {

            PDFRendererrenderer = newPDFRenderer(document);

            intpageCount = document.getNumberOfPages();

          

            StringimageDir = previewCachePath +"/images/"+System.currentTimeMillis() +"/";

            FileUtil.mkdir(imageDir);

          

            for (inti = 0; i < pageCount; i++) {

                BufferedImageimage = renderer.renderImageWithDPI(i,200,ImageType.RGB);

                StringimagePath = imageDir +"page_"+ (i +1) +".jpg";

                ImageIO.write(image,"JPEG",newFile(imagePath));

                imagePaths.add(imagePath);

            }

          

            log.info("PDF转图片完成，共{}页", pageCount);

        }

      

        return imagePaths;

    }

  

    /**

     * Office文档转PDF（用于在线预览）

     */

    publicString convertOfficeToPdf(StringofficePath){

        StringfileExtension = FileUtil.extName(officePath);

      

        // 支持的格式：doc, docx, xls, xlsx, ppt, pptx

        if (!StrUtil.equalsAnyIgnoreCase(fileExtension,

            "doc","docx","xls","xlsx","ppt","pptx")) {

            thrownewRuntimeException("不支持的文件格式："+ fileExtension);

        }

      

        // 调用转换服务

        returnconvertService.convertToPdf(officePath);

    }

  

    /**

     * 获取文档页数

     */

    publicint getDocumentPageCount(StringfilePath)throwsIOException{

        Stringextension = FileUtil.extName(filePath);

      

        if ("pdf".equalsIgnoreCase(extension)) {

            try (PDDocumentdocument = PDDocument.load(newFile(filePath))) {

                returndocument.getNumberOfPages();

            }

        }

      

        // 其他格式先转PDF再统计

        StringpdfPath = convertService.convertToPdf(filePath);

        try (PDDocumentdocument = PDDocument.load(newFile(pdfPath))) {

            returndocument.getNumberOfPages();

        }

    }

  

    /**

     * 提取PDF文本

     */

    publicString extractPdfText(StringpdfPath)throwsIOException{

        try (PDDocumentdocument = PDDocument.load(newFile(pdfPath))) {

            org.apache.pdfbox.text.PDFTextStripperstripper = 

                new org.apache.pdfbox.text.PDFTextStripper();

            returnstripper.getText(document);

        }

    }

}
```

```java
        } catch (OfficeExceptione) {
```

```java
packagecom.clm.system.document.service;


importlombok.extern.slf4j.Slf4j;

importnet.sourceforge.tess4j.Tesseract;

importnet.sourceforge.tess4j.TesseractException;

importorg.springframework.beans.factory.annotation.Value;

importorg.springframework.stereotype.Service;


importjavax.annotation.PostConstruct;

importjava.io.File;


/**

 * OCR 文字识别服务

 */

@Slf4j

@Service

publicclassDocumentOcrService {

  

    @Value("${document.ocr.tessdata-path:D:/tessdata}")

    privateStringtessdataPath;

  

    @Value("${document.ocr.languages:chi_sim+eng}")

    privateStringlanguages;

  

    privateTesseracttesseract;

  

    @PostConstruct

    publicvoid init(){

        tesseract = newTesseract();

      

        // 设置训练数据路径

        tesseract.setDatapath(tessdataPath);

      

        // 设置语言（chi_sim：简体中文，eng：英文）

        tesseract.setLanguage(languages);

      

        // 设置识别引擎模式（3：默认，基于LSTM）

        tesseract.setOcrEngineMode(3);

      

        // 设置页面分割模式（3：自动分割）

        tesseract.setPageSegMode(3);

      

        log.info("OCR服务初始化完成，语言：{}", languages);

    }

  

    /**

     * 识别图片中的文字

     */

    publicString recognizeImage(StringimagePath){

        try {

            FileimageFile = newFile(imagePath);

            Stringresult = tesseract.doOCR(imageFile);

            log.info("OCR识别成功，文件：{}，文字长度：{}", imagePath,result.length());

            return result;

        } catch (TesseractExceptione) {

            log.error("OCR识别失败：{}", imagePath, e);

            thrownewRuntimeException("OCR识别失败："+e.getMessage());

        }

    }

  

    /**

     * 识别PDF中的文字（逐页识别）

     */

    publicString recognizePdf(StringpdfPath,DocumentPreviewServicepreviewService){

        try {

            // 1. PDF转图片

            varimagePaths = previewService.convertPdfToImages(pdfPath);

          

            // 2. 逐页OCR识别

            StringBuilderresult = newStringBuilder();

            for (inti = 0; i <imagePaths.size(); i++) {

                Stringtext = recognizeImage(imagePaths.get(i));

                result.append("--- 第 ").append(i +1).append(" 页 ---\n");

                result.append(text).append("\n\n");

            }

          

            returnresult.toString();

        } catch (Exceptione) {

            log.error("PDF OCR识别失败", e);

            thrownewRuntimeException("PDF OCR识别失败："+e.getMessage());

        }

    }

  

    /**

     * 识别指定区域的文字

     */

    publicString recognizeRegion(StringimagePath,intx,inty,intwidth,intheight){

        try {

            FileimageFile = newFile(imagePath);

            java.awt.Rectanglerect = new java.awt.Rectangle(x, y, width, height);

            Stringresult = tesseract.doOCR(imageFile, rect);

            return result;

        } catch (TesseractExceptione) {

            log.error("OCR区域识别失败", e);

            thrownewRuntimeException("OCR区域识别失败："+e.getMessage());

        }

    }

}
```

```java
importcom.fasterxml.jackson.databind.ObjectMapper;

/**

@Slf4j

@Component

@RequiredArgsConstructor

publicclassDocumentCollaborateHandlerextendsTextWebSocketHandler {

  

    privatefinalObjectMapperobjectMapper;

  

    // 文件ID -> 用户会话映射

    privatefinalMap<Long,Map<String,WebSocketSession>> fileSessions = newConcurrentHashMap<>();

  

    // 文件编辑锁

    privatefinalMap<Long,String> fileLocks = newConcurrentHashMap<>();

  

    @Override

    publicvoid afterConnectionEstablished(WebSocketSessionsession)throwsException{

        LongfileId = getFileId(session);

        StringuserId = getUserId(session);

        StringuserName = getUserName(session);

      

        // 加入协同编辑房间

        fileSessions.computeIfAbsent(fileId, k ->newConcurrentHashMap<>())

            .put(userId, session);

      

        // 获取当前在线用户列表

        Map<String,WebSocketSession> sessions = fileSessions.get(fileId);

      

        // 广播：新用户加入

        broadcastMessage(fileId,Map.of(

            "type","user_join",

            "userId", userId,

            "userName", userName,

            "onlineUsers",sessions.size(),

            "timestamp",System.currentTimeMillis()

        ));

      

        // 发送当前编辑锁状态

        StringlockOwner = fileLocks.get(fileId);

        if (lockOwner !=null) {

            session.sendMessage(newTextMessage(objectMapper.writeValueAsString(Map.of(

                "type","lock_status",

                "locked",true,

                "lockOwner", lockOwner

            ))));

        }

      

        log.info("用户加入文档编辑，文件ID：{}，用户：{}，当前在线：{}", fileId, userName,sessions.size());

    }

  

    @Override

    protectedvoid handleTextMessage(WebSocketSessionsession,TextMessagemessage)throwsException{

        LongfileId = getFileId(session);

        StringuserId = getUserId(session);

      

        // 解析消息

        @SuppressWarnings("unchecked")

        Map<String,Object> data = objectMapper.readValue(message.getPayload(),Map.class);

        Stringtype = (String) data.get("type");

      

        switch (type) {

            case"edit":

                // 编辑操作：插入/删除文字

                handleEdit(fileId, userId, data);

                break;

              

            case"cursor":

                // 光标移动

                handleCursorMove(fileId, userId, data);

                break;

              

            case"save":

                // 保存文档

                handleSave(fileId, userId, data);

                break;

              

            case"lock":

                // 请求编辑锁

                handleLockRequest(fileId, userId, session);

                break;

              

            case"unlock":

                // 释放编辑锁

                handleUnlock(fileId, userId);

                break;

        }

    }

  

    @Override

    publicvoid afterConnectionClosed(WebSocketSessionsession,CloseStatusstatus)throwsException{

        LongfileId = getFileId(session);

        StringuserId = getUserId(session);

        StringuserName = getUserName(session);

      

        // 移除会话

        Map<String,WebSocketSession> sessions = fileSessions.get(fileId);

        if (sessions !=null) {

            sessions.remove(userId);

          

            // 如果用户持有编辑锁，释放锁

            if (userId.equals(fileLocks.get(fileId))) {

                fileLocks.remove(fileId);

                broadcastMessage(fileId,Map.of(

                    "type","lock_released",

                    "userId", userId

                ));

            }

          

            // 广播：用户离开

            broadcastMessage(fileId,Map.of(

                "type","user_leave",

                "userId", userId,

                "userName", userName,

                "onlineUsers",sessions.size()

            ));

          

            // 如果没有用户了，清理房间

            if (sessions.isEmpty()) {

                fileSessions.remove(fileId);

            }

        }

      

        log.info("用户离开文档编辑，文件ID：{}，用户：{}", fileId, userName);

    }

            // 锁可用，分配给请求者

            session.sendMessage(newTextMessage(objectMapper.writeValueAsString(Map.of(

        } else {

                "type","lock_released",

                "userId", userId
```

### 5.3 文件秒传

```vue

<template>

  <div>

    <a-upload

      :before-upload="beforeUpload"

      :custom-request="handleUpload"

    >

      <a-button>

        <upload-outlined></upload-outlined>

        选择文件

      </a-button>

    </a-upload>

  </div>

</template>


<scriptsetup>

import { message } from'ant-design-vue'

importSparkMD5from'spark-md5'

import { checkQuickUpload, quickUpload, uploadFile } from'@/api/document'


asyncfunctionbeforeUpload(file) {

  // 计算文件MD5

  constmd5 = awaitcalculateFileMD5(file)

  file.md5 = md5

  returntrue

}


asyncfunctionhandleUpload({ file }) {

  constmd5 = file.md5

  

  // 1. 检查文件是否存在（秒传）

  const { data } = awaitcheckQuickUpload(md5)

  

  if (data.exists) {

    // 文件已存在，执行秒传

    awaitquickUpload({

      fileHash:md5,

      fileName:file.name,

      fileSize:file.size,

      fileType:file.type

    })

    message.success('文件秒传成功')

  } else {

    // 文件不存在，正常上传

    awaituploadFile(file)

    message.success('文件上传成功')

  }

}


functioncalculateFileMD5(file) {

  returnnewPromise((resolve, reject) => {

    constblobSlice = File.prototype.slice

    constchunkSize = 2097152// 2MB

    constchunks = Math.ceil(file.size/chunkSize)

    letcurrentChunk = 0

    constspark = newSparkMD5.ArrayBuffer()

    constfileReader = newFileReader()

  

    fileReader.onload = (e) => {

      spark.append(e.target.result)

      currentChunk++

    

      if (currentChunk < chunks) {

        loadNext()

      } else {

        constmd5 = spark.end()

        resolve(md5)

      }

    }

  

    fileReader.onerror = () => {

      reject('文件读取失败')

    }

  

    functionloadNext() {

      conststart = currentChunk*chunkSize

      constend = Math.min(start+chunkSize, file.size)

      fileReader.readAsArrayBuffer(blobSlice.call(file, start, end))

    }

  

    loadNext()

  })

}

</script>
```
