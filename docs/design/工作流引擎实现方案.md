# 工作流引擎（Flowable）实现方案

## 一、技术架构

### 1.1 核心依赖配置

在父 `pom.xml` 添加版本管理：

```xml
<properties>

    <bpmn-js.version>9.0.3</bpmn-js.version>
</properties>

<dependencyManagement>
    <dependencies>
        <!-- Flowable 工作流引擎 -->
        <dependency>
            <groupId>org.flowable</groupId>

            <version>${flowable.version}</version>
        </dependency>
        
        <!-- Flowable UI 模块化（可选） -->
        <dependency>

    </dependencies>
```


### 2.2 Flowable 自动创建的表

Flowable 启动时会自动创建约 25 张表，主要分为以下几类：

**流程定义（Repository）：**
- `ACT_RE_DEPLOYMENT`: 部署表
- `ACT_RE_PROCDEF`: 流程定义表
- `ACT_RE_MODEL`: 流程模型表

**运行时数据（Runtime）：**
- `ACT_RU_EXECUTION`: 流程执行表
- `ACT_RU_TASK`: 任务表
- `ACT_RU_VARIABLE`: 变量表
- `ACT_RU_EVENT_SUBSCR`: 事件订阅表

**历史数据（History）：**
- `ACT_HI_PROCINST`: 历史流程实例表
- `ACT_HI_TASKINST`: 历史任务表
- `ACT_HI_ACTINST`: 历史活动表
- `ACT_HI_VARINST`: 历史变量表
- `ACT_HI_COMMENT`: 历史意见表

**身份信息（Identity）：**
- `ACT_ID_USER`: 用户表
- `ACT_ID_GROUP`: 组表
- `ACT_ID_MEMBERSHIP`: 用户组关系表

### 3.1 Flowable 配置类

```java
package com.clm.workflow.config;

import org.flowable.spring.SpringProcessEngineConfiguration;
import org.flowable.spring.boot.EngineConfigurationConfigurer;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

/**
 * Flowable 配置
 */
@Configuration
public class FlowableConfig {
    
    /**
     * 流程引擎配置
     */
    @Bean
    public EngineConfigurationConfigurer<SpringProcessEngineConfiguration> processEngineConfigurer() {
        return engineConfiguration -> {
            // 设置字体（解决流程图中文乱码）
            engineConfiguration.setActivityFontName("宋体");
            engineConfiguration.setLabelFontName("宋体");
            engineConfiguration.setAnnotationFontName("宋体");
            
            // 开启历史记录（full：记录所有）
            engineConfiguration.setHistoryLevel(
                org.flowable.engine.impl.history.HistoryLevel.FULL
            );
            
            // 禁用定时任务JOB（如果使用Quartz统一调度）
            // engineConfiguration.setAsyncExecutorActivate(false);
            
            // 自动部署流程定义
            engineConfiguration.setDeploymentMode("default");
        };
    }
}
```

### 3.3 核心Service

```java
package com.clm.workflow.service;

import cn.hutool.core.util.StrUtil;
import cn.hutool.json.JSONUtil;
import com.clm.workflow.domain.dto.*;
import com.clm.workflow.domain.entity.WorkflowInstance;
import com.clm.workflow.domain.entity.WorkflowHistory;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.flowable.bpmn.model.BpmnModel;
import org.flowable.engine.*;
import org.flowable.engine.history.*;
import org.flowable.engine.repository.ProcessDefinition;
import org.flowable.engine.runtime.ProcessInstance;
import org.flowable.image.ProcessDiagramGenerator;
import org.flowable.task.api.Task;
import org.flowable.task.api.history.HistoricTaskInstance;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.io.InputStream;
import java.util.*;
import java.util.stream.Collectors;

/**
 * 工作流服务
 */
@Slf4j
@Service
@RequiredArgsConstructor
public class WorkflowService {
    
    private final RuntimeService runtimeService;
    private final TaskService taskService;
    private final HistoryService historyService;
    private final RepositoryService repositoryService;
    private final IdentityService identityService;
    private final ProcessEngine processEngine;
    
    /**
     * 启动流程
     */
    @Transactional(rollbackFor = Exception.class)
    public String startProcess(ProcessStartDTO dto, String userId, String userName) {
        // 设置流程发起人
        identityService.setAuthenticatedUserId(userId);
        
        // 启动流程
        ProcessInstance processInstance = runtimeService.startProcessInstanceByKey(
            dto.getProcessKey(),
            dto.getBusinessKey(),
            dto.getVariables()
        );
        
        // 保存流程实例扩展信息
        WorkflowInstance instance = new WorkflowInstance();
        instance.setProcessInstanceId(processInstance.getId());
        instance.setProcessDefinitionId(processInstance.getProcessDefinitionId());
        instance.setProcessName(processInstance.getProcessDefinitionName());
        instance.setBusinessKey(dto.getBusinessKey());
        instance.setBusinessType(dto.getBusinessType());
        instance.setTitle(dto.getTitle());
        instance.setInitiator(userId);
        instance.setInitiatorName(userName);
        instance.setFormData(JSONUtil.toJsonStr(dto.getVariables()));
        instance.setStatus(1); // 进行中
        instance.setPriority(50); // 默认中等优先级
        // workflowInstanceMapper.insert(instance);
        
        log.info("流程启动成功，实例ID：{}，标题：{}", processInstance.getId(), dto.getTitle());
        return processInstance.getId();
    }
    
    /**
     * 查询待办任务
     */
    public List<Map<String, Object>> queryTodoTasks(String userId) {
        List<Task> tasks = taskService.createTaskQuery()
            .taskCandidateOrAssigned(userId)
            .orderByTaskCreateTime()
            .desc()
            .list();
        
        return tasks.stream().map(task -> {
            Map<String, Object> map = new HashMap<>();
            map.put("taskId", task.getId());
            map.put("taskName", task.getName());
            map.put("taskKey", task.getTaskDefinitionKey());
            map.put("processInstanceId", task.getProcessInstanceId());
            map.put("processDefinitionId", task.getProcessDefinitionId());
            map.put("assignee", task.getAssignee());
            map.put("createTime", task.getCreateTime());
            map.put("dueDate", task.getDueDate());
            map.put("priority", task.getPriority());
            
            // 获取流程变量
            Map<String, Object> variables = taskService.getVariables(task.getId());
            map.put("variables", variables);
            
            // 获取流程定义信息
            ProcessDefinition processDefinition = repositoryService
                .createProcessDefinitionQuery()
                .processDefinitionId(task.getProcessDefinitionId())
                .singleResult();
            map.put("processName", processDefinition.getName());
            
            return map;
        }).collect(Collectors.toList());
    }
    
    /**
     * 查询已办任务
     */
    public List<Map<String, Object>> queryDoneTasks(String userId) {
        List<HistoricTaskInstance> tasks = historyService.createHistoricTaskInstanceQuery()
            .taskAssignee(userId)
            .finished()
            .orderByHistoricTaskInstanceEndTime()
            .desc()
            .list();
        
        return tasks.stream().map(task -> {
            Map<String, Object> map = new HashMap<>();
            map.put("taskId", task.getId());
            map.put("taskName", task.getName());
            map.put("processInstanceId", task.getProcessInstanceId());
            map.put("startTime", task.getCreateTime());
            map.put("endTime", task.getEndTime());
            map.put("duration", task.getDurationInMillis());
            
            // 获取审批意见
            List<org.flowable.engine.task.Comment> comments = taskService.getTaskComments(task.getId());
            if (!comments.isEmpty()) {
                map.put("comment", comments.get(0).getFullMessage());
            }
            
            return map;
        }).collect(Collectors.toList());
    }
    
    /**
     * 完成任务（审批）
     */
    @Transactional(rollbackFor = Exception.class)
    public void completeTask(TaskCompleteDTO dto, String userId, String userName) {
        String taskId = dto.getTaskId();
        
        // 添加审批意见
        if (StrUtil.isNotBlank(dto.getComment())) {
            taskService.addComment(taskId, null, dto.getComment());
        }
        
        // 设置审批结果到流程变量
        Map<String, Object> variables = dto.getVariables();
        if (variables == null) {
            variables = new HashMap<>();
        }
        variables.put("approved", "agree".equals(dto.getAction()));
        variables.put("approver", userId);
        variables.put("approverName", userName);
        
        // 完成任务
        taskService.complete(taskId, variables);
        
        // 保存历史记录
        Task task = taskService.createTaskQuery().taskId(taskId).singleResult();
        if (task != null) {
            WorkflowHistory history = new WorkflowHistory();
            history.setProcessInstanceId(task.getProcessInstanceId());
            history.setTaskId(taskId);
            history.setTaskName(task.getName());
            history.setAssignee(userId);
            history.setAssigneeName(userName);
            history.setAction(dto.getAction());
            history.setComment(dto.getComment());
            // workflowHistoryMapper.insert(history);
        }
        
        log.info("任务完成，任务ID：{}，处理人：{}，结果：{}", taskId, userName, dto.getAction());
    }
    
    /**
     * 签收任务
     */
    @Transactional(rollbackFor = Exception.class)
    public void claimTask(String taskId, String userId) {
        taskService.claim(taskId, userId);
        log.info("任务签收成功，任务ID：{}，签收人：{}", taskId, userId);
    }
    
    /**
     * 转办任务
     */
    @Transactional(rollbackFor = Exception.class)
    public void transferTask(TaskTransferDTO dto, String userId) {
        String taskId = dto.getTaskId();
        
        // 添加转办意见
        if (StrUtil.isNotBlank(dto.getComment())) {
            taskService.addComment(taskId, null, "转办：" + dto.getComment());
        }
        
        // 转办任务
        taskService.setAssignee(taskId, dto.getTargetUserId());
        
        log.info("任务转办成功，任务ID：{}，转办人：{}，接收人：{}", 
            taskId, userId, dto.getTargetUserId());
    }
    
    /**
     * 委托任务
     */
    @Transactional(rollbackFor = Exception.class)
    public void delegateTask(TaskDelegateDTO dto, String userId) {
        String taskId = dto.getTaskId();
        
        // 添加委托意见
        if (StrUtil.isNotBlank(dto.getComment())) {
            taskService.addComment(taskId, null, "委托：" + dto.getComment());
        }
        
        // 委托任务
        taskService.delegateTask(taskId, dto.getDelegateUserId());
        
        log.info("任务委托成功，任务ID：{}，委托人：{}，被委托人：{}", 
            taskId, userId, dto.getDelegateUserId());
    }
    
    /**
     * 获取流程图（高亮已完成节点）
     */
    public InputStream getProcessDiagram(String processInstanceId) {
        // 获取历史流程实例
        HistoricProcessInstance historicProcessInstance = 
            historyService.createHistoricProcessInstanceQuery()
                .processInstanceId(processInstanceId)
                .singleResult();
        
        if (historicProcessInstance == null) {
            return null;
        }
        
        // 获取流程定义
        String processDefinitionId = historicProcessInstance.getProcessDefinitionId();
        BpmnModel bpmnModel = repositoryService.getBpmnModel(processDefinitionId);
        
        // 获取已完成的活动ID
        List<HistoricActivityInstance> historicActivityList = 
            historyService.createHistoricActivityInstanceQuery()
                .processInstanceId(processInstanceId)
                .orderByHistoricActivityInstanceId()
                .asc()
                .list();
        
        List<String> highLightedActivities = historicActivityList.stream()
            .filter(activity -> activity.getEndTime() != null)
            .map(HistoricActivityInstance::getActivityId)
            .distinct()
            .collect(Collectors.toList());
        
        // 获取流程图生成器
        ProcessDiagramGenerator diagramGenerator = processEngine.getProcessEngineConfiguration()
            .getProcessDiagramGenerator();
        
        // 生成流程图
        return diagramGenerator.generateDiagram(
            bpmnModel, 
            "png",
            highLightedActivities,
            Collections.emptyList(),
            "宋体", 
            "宋体", 
            "宋体",
            null,
            1.0,
            false
        );
    }
    
    /**
     * 撤销流程
     */
    @Transactional(rollbackFor = Exception.class)
    public void cancelProcess(String processInstanceId, String reason, String userId) {
        // 删除流程实例
        runtimeService.deleteProcessInstance(processInstanceId, reason);
        
        // 更新扩展表状态
        // WorkflowInstance instance = workflowInstanceMapper.selectByProcessInstanceId(processInstanceId);
        // instance.setStatus(4); // 已撤销
        // workflowInstanceMapper.updateById(instance);
        
        log.info("流程已撤销，实例ID：{}，撤销人：{}，原因：{}", processInstanceId, userId, reason);
    }
    
    /**
     * 获取流程历史记录
     */
    public List<Map<String, Object>> getProcessHistory(String processInstanceId) {
        List<HistoricActivityInstance> list = historyService.createHistoricActivityInstanceQuery()
            .processInstanceId(processInstanceId)
            .orderByHistoricActivityInstanceStartTime()
            .asc()
            .list();
        
        return list.stream().map(activity -> {
            Map<String, Object> map = new HashMap<>();
            map.put("activityId", activity.getActivityId());
            map.put("activityName", activity.getActivityName());
            map.put("activityType", activity.getActivityType());
            map.put("assignee", activity.getAssignee());
            map.put("startTime", activity.getStartTime());
            map.put("endTime", activity.getEndTime());
            map.put("duration", activity.getDurationInMillis());
            
            // 获取审批意见
            if ("userTask".equals(activity.getActivityType()) && activity.getTaskId() != null) {
                List<org.flowable.engine.task.Comment> comments = 
                    taskService.getTaskComments(activity.getTaskId());
                if (!comments.isEmpty()) {
                    map.put("comment", comments.get(0).getFullMessage());
                }
            }
            
            return map;
        }).collect(Collectors.toList());
    }
}
```
